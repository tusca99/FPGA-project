\section{Percolation Theory}
\begin{frame}{What is Percolation?}
  \textbf{Physical picture:} a fluid filters through a porous medium.

  \begin{columns}
    \column{0.45\textwidth}
    \begin{itemize}
      \item 2D grid: each site is \alert{open} (blue, prob. $p$) or \alert{closed} (gray, $1-p$)
      \item A \alert{spanning cluster} connects top $\to$ bottom
      \item \alert{Phase transition} at critical probability $\pc$
      \item Below $\pc$: only finite clusters \\
            Above $\pc$: spanning cluster emerges
      \item \textbf{Isotropic percolation:} fluid can move in all 4 directions
    \end{itemize}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=0.6\textwidth]{percolation_isotropic.png}
  \end{columns}
\end{frame}

\begin{frame}{Directed Percolation (DP)}
  \textbf{Key difference:} flow is \alert{directional} — only downward propagation.

  \begin{columns}
    \column{0.45\textwidth}
    \begin{itemize}
      \item Fluid moves only $\downarrow$ (not $\uparrow$)
      \item Horizontal spread still allowed ($\leftrightarrow$)
      \item Different universality class from isotropic percolation
      \item DP critical exponents:
        {\small
        \begin{align*}
          \beta &= 0.2765 \quad &&\text{(order parameter)}\\
          \nu_{\|} &= 1.096   \quad &&\text{(corr.\ length, time)}\\
          \nu_{\perp} &= 0.733 \quad &&\text{(corr.\ length, space)}
        \end{align*}
        }
    \end{itemize}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=0.6\textwidth]{percolation_directed.png}
  \end{columns}
\end{frame}

\begin{frame}{Finite-Size Scaling}
  For a finite grid of size $N \times N$, the spanning probability obeys:
  \[
  P_{\text{span}}(p, N) = F\!\left((p - \pc)\, N^{\nup}\right)
  \]
  where $F$ is a \alert{universal scaling function}.

  \vspace{0.3cm}
  \textbf{Consequences:}
  \begin{itemize}
    \item Curves for different $N$ \alert{collapse} onto a single master curve
    \item The collapse validates the universality class
    \item The crossing point of $P_{\text{span}} = 0.5$ gives an estimate of $\pc$
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Threshold estimation} via logistic regression:
  \[
  P_{\text{span}}(p) = \frac{1}{1 + e^{-a(p - \pc)}}
  \]
  Fitted via maximum likelihood (binomial bootstrap for CIs).
\end{frame}

\begin{frame}{RNG Bank — $N_{\text{rows}}\times$ Trivium Stream Cipher}
  \begin{itemize}
    \item \alert{$N_{\text{rows}}$ independent} Trivium stream ciphers, one per grid column
    \item $N_{\text{rows}}$ is a \textbf{compile-time generic} (\texttt{N\_ROWS\_G}), default 64
    \item Synthesized for 64, 128, or 180 — same throughput, different resource usage
    \item \textbf{AES-CTR seeding} (one-time, $2 \times N_{\text{rows}} \times 10$ cycles):
      \begin{itemize}
        \item $2 \times N_{\text{rows}}$ AES blocks $\times$ 10 cycles/block
        \item For $N_{\text{rows}}=64$: 128 $\times$ 10 = 1280 cycles
      \end{itemize}
    \item \textbf{Trivium warmup}: 1152 bit-cycles / 32 bit/cycle = 36 cycles
    \item \textbf{Total warmup} ($N_{\text{rows}}=64$): $\approx$1316 cycles $\approx$ 13.2 $\mu$s at 100 MHz
  \end{itemize}

  \vspace{0.15cm}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Per cycle:}
    \begin{itemize}
      \item $N_{\text{rows}}$ random 32-bit words
      \item Compared to threshold $p$ (UQ32 fixed-point)
      \item Output: $N_{\text{rows}}$-bit \texttt{site\_open} vector
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Validation:}
    \begin{itemize}
      \item Occupancy bias $< 0.001$
      \item Uniform distribution verified
      \item No correlation between columns
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Frontier Engine — Row-Wise Reachability}
  \textbf{Problem:} determine if a top-to-bottom path exists without storing the full grid.

  \vspace{0.2cm}
  \textbf{Algorithm (per row):}
  \begin{enumerate}
    \item \alert{Vertical seed:}
      \[
      \text{seed}[i] = \text{open}[i] \land \text{prev\_reach}[i]
      \]
      (Row 0: $\text{seed} = \text{open}$)
    \item \alert{Horizontal closure} (iterative $\pm1$ expansion):
      \[
      \text{reach}[i] = \text{seed}[i] \lor \bigl(\text{open}[i] \land (\text{reach}[i{-}1] \lor \text{reach}[i{+}1])\bigr)
      \]
      Converges in $\leq N_{\text{rows}}$ iterations (typically 2--3 for $N=64$)
  \end{enumerate}

  \vspace{0.2cm}
  \textbf{VHDL implementation:} bidirectional associative prefix scan
  \begin{itemize}
    \item \alert{3-stage pipeline}: 3 cycles/row (READY $\to$ COMPUTE $\to$ SAVE)
    \item \alert{4 cycles/row end-to-end}: the core's registered row-send handshake adds 1 cycle/row
    \item Only 2 rows of state (current + previous reachability)
    \item Validated against BFS: 1000 random tests, exact match
  \end{itemize}
\end{frame}

\section{FPGA Engineering Results}

\begin{frame}{Latency vs Batch Size — Amdahl Speedup}
  \centering
  \includegraphics[width=0.75\textwidth]{output/analysis/latency_vs_batch.png}
  \vspace{0.2cm}
  \begin{itemize}
    \item Fixed UART overhead $\approx$2.78 ms dominates at small $R$
    \item Speedup follows Amdahl's law: serial fraction $f_{\text{serial}} \approx 0.2$--$0.4$
    \item Break-even batch size: $R \approx 100$--$1000$ depending on grid height
  \end{itemize}
\end{frame}

\begin{frame}{Asymptotic Cycles-Per-Run Decomposition}
  \centering
  \includegraphics[width=0.55\textwidth]{output/analysis/breakdown_fit.png}

  \textbf{Model:} $T_{\text{total}} \times R = C_{\text{fixed}} + C_{\text{marginal}} \times R$
  \begin{itemize}
    \item $C_{\text{marginal}}$: per-run core cost (asymptote at large $R$)
    \item $C_{\text{fixed}}$: UART + host overhead (one-time per request)
    \item Asymptote matches theoretical $4 \times S$ cycles/run (3 frontier + 1 handshake)
  \end{itemize}
\end{frame}

\begin{frame}{Pipeline Efficiency}
  \centering
  \includegraphics[width=0.75\textwidth]{output/analysis/pipeline_efficiency.png}
  \vspace{0.2cm}
  \begin{itemize}
    \item Measured: $C_{\text{core}} = 3.99 \times S + \text{const}$ (vs ideal $4 \times S$)
    \item \alert{Efficiency: $\approx$100\%} of the achievable $4 \times S$ model
    \item The $4^{\text{th}}$ cycle/row is the core's \alert{registered-send handshake}, not a frontier bubble
  \end{itemize}
\end{frame}

\begin{frame}{Grid-Height Invariance \& Determinism}
  \begin{columns}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=\textwidth]{output/analysis/throughput_invariance.png}\\
    \small Throughput vs grid height
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=\textwidth]{output/analysis/determinism_cv.png}\\
    \small Coefficient of variation
  \end{columns}
  \vspace{0.3cm}
  \begin{itemize}
    \item Throughput \alert{independent} of grid height (pipeline-dominated)
    \item CV $< 1\%$ across all configurations — FPGA is deterministic

  \end{itemize}
\end{frame}
